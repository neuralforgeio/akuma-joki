    55→function formatTime(ts: number) {
    56→  return new Date(ts).toLocaleString("id-ID", { day: "2-digit", month: "short", hour: "2-digit", minute: "2-digit" });
    57→}
    58→
    59→function formatRelative(ts: number) {
    60→  const diff = Date.now() - ts;
    61→  if (diff < 60000) return "just now";
    62→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
    63→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`;
    64→  return `${Math.floor(diff / 86400000)}d ago`;
    65→}
    66→
    67→function formatDuration(ms: number | null) {
    68→  if (!ms) return "-";
    69→  if (ms < 1000) return `${ms}ms`;
    70→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`;
    71→  return `${Math.floor(ms / 60000)}m ${Math.floor((ms % 60000) / 1000)}s`;
    72→}

   643→            </div>

   758→function InfoRow({ icon: Icon, label, value, mono, badge, avatar }: { icon: typeof Code; label: string; value: string; mono?: boolean; badge?: boolean; avatar?: string | null }) {
   759→  return (
   760→    <div className="flex items-center gap-3 py-1.5 border-b border-white/5 last:border-0">
   761→      <Icon className="size-3.5 text-zinc-500 shrink-0" />
   762→      <span className="text-[11px] text-zinc-500 uppercase tracking-wider w-20 shrink-0">{label}</span>
   763→      <div className="flex-1 min-w-0 flex items-center gap-2">
   764→        {avatar && <img src={avatar} alt="" className="size-4 rounded-full" />}
   765→        <span className={cn("text-sm text-zinc-100 truncate", mono && "font-mono text-cyan-400")}>
   766→          {badge ? <span className="px-1.5 py-0.5 rounded bg-violet-500/10 text-violet-400 text-[10px] uppercase">{value}</span> : value}
   767→        </span>
   768→      </div>
   769→    </div>
   770→  );
   771→}
   772→
   773→function InfoCard({ label, value, link, mono }: { label: string; value: string; link?: boolean | string; mono?: boolean }) {
   774→  const isLink = link === true || typeof link === "string";
   775→  const href = typeof link === "string" ? link : value;
   776→  return (
   777→    <div className="glass rounded-xl p-3">
   778→      <p className="text-[10px] uppercase text-zinc-500 tracking-wider mb-1">{label}</p>
   779→      {isLink ? (
   780→        <a href={href} target="_blank" rel="noopener noreferrer" className={cn("text-sm text-cyan-400 hover:underline truncate block", mono && "font-mono")}>
   781→          {value}
   782→        </a>
   783→      ) : (
   784→        <p className={cn("text-sm text-zinc-100 truncate", mono && "font-mono")}>{value}</p>
   785→      )}
   786→    </div>
   787→  );
   788→}
