     1→"use client";
     2→
     3→import { useState, useMemo, useEffect } from "react";
     4→import { useRouter } from "next/navigation";
     5→import Link from "next/link";
     6→import { ArrowLeft, ShoppingCart, Tag, Check, AlertTriangle, Lock, Star, X, Plus, Minus } from "lucide-react";
     7→import type { Game } from "@/lib/games-data";
     8→import type { ProductItem, ProductCategory } from "@/lib/games-data";
     9→import { GAMES } from "@/lib/games-data";
    10→import { useAkumaStore } from "@/lib/store";
    11→import { useAdminStore } from "@/lib/admin-store";
    12→import { useReviews } from "@/lib/reviews";
    13→import { useRecentlyViewed } from "@/lib/recently-viewed";
    14→import { useCart, MAX_CART_ITEMS } from "@/lib/cart";
    15→import { useToast } from "@/hooks/use-toast";
    16→import { PixelButton } from "./pixel-button";
    17→import { Reveal } from "./reveal";
    18→import { SkeletonGrid } from "./skeleton";
    19→import { WishlistButton } from "./wishlist-button";
    20→import { cn } from "@/lib/utils";
    21→import { AnimatePresence, motion } from "framer-motion";
    22→
    23→export function StoreView({ game }: { game: Game }) {
    24→  const router = useRouter();
    25→  const order = useAkumaStore((s) => s.order);
    26→  const selectProduct = useAkumaStore((s) => s.selectProduct);
    27→  const cartAdd = useCart((s) => s.add);
    28→  const cartHas = useCart((s) => s.has);
    29→  const cartCount = useCart((s) => s.items.length);
    30→  const addViewed = useRecentlyViewed((s) => s.addViewed);
    31→  const trackGameView = useAdminStore((s) => s.trackGameView);
    32→  const { toast } = useToast();
    33→  const [selectedItem, setSelectedItem] = useState<{ item: ProductItem; category: string } | null>(null);
    34→  const [cartLimitModal, setCartLimitModal] = useState(false);
    35→
    36→  // Track game view on mount
    37→  useEffect(() => {
    38→    trackGameView(game.slug);
    39→  }, [game.slug, trackGameView]);
    40→
    41→  /* ===== Advanced Filter State ===== */
    42→  const [filterQuery, setFilterQuery] = useState("");
    43→  const [filterCategory, setFilterCategory] = useState<string>("all");
    44→  const [filterTag, setFilterTag] = useState<string>("all");
    45→  const [sortBy, setSortBy] = useState<"default" | "price-asc" | "price-desc" | "name">("default");
    46→
    47→  // Collect all unique tags from items
    48→  const allTags = useMemo(() => {
    49→    const tags = new Set<string>();
    50→    game.categories.forEach((c) => c.items.forEach((i) => { if (i.tag) tags.add(i.tag); }));
    51→    return Array.from(tags);
    52→  }, [game]);
    53→
    54→  // Filter + sort items
    55→  const filteredCategories = useMemo(() => {
    56→    const q = filterQuery.trim().toLowerCase();
    57→    let result = game.categories.map((cat) => ({
    58→      ...cat,
    59→      items: cat.items.filter((item) => {
    60→        if (filterCategory !== "all" && cat.id !== filterCategory) return false;
    61→        if (filterTag !== "all" && item.tag !== filterTag) return false;
    62→        if (q && !item.name.toLowerCase().includes(q) && !(item.description || "").toLowerCase().includes(q)) return false;
    63→        return true;
    64→      }),
    65→    }));
    66→    // Filter out empty categories when search/filter active
    67→    if (q || filterTag !== "all") {
    68→      result = result.filter((c) => c.items.length > 0);
    69→    }
    70→    // Sort items within each category
    71→    if (sortBy !== "default") {
    72→      result = result.map((c) => ({
    73→        ...c,
    74→        items: [...c.items].sort((a, b) => {
    75→          if (sortBy === "price-asc") return a.price - b.price;
    76→          if (sortBy === "price-desc") return b.price - a.price;
    77→          if (sortBy === "name") return a.name.localeCompare(b.name);
    78→          return 0;
    79→        }),
    80→      }));
    81→    }
    82→    // Hide categories not matching filterCategory
    83→    if (filterCategory !== "all") {
    84→      result = result.filter((c) => c.id === filterCategory);
    85→    }
    86→    return result;
    87→  }, [game, filterQuery, filterCategory, filterTag, sortBy]);
    88→
    89→  const handlePick = (
    90→    item: (typeof game.categories)[number]["items"][number],
    91→    categoryName: string
    92→  ) => {
    93→    const pItem = item as ProductItem;
    94→    // Track ke recently viewed
    95→    addViewed({
    96→      id: `${game.slug}-${pItem.id}`,
    97→      gameSlug: game.slug,
    98→      gameName: game.name,
    99→      gameEmoji: game.emoji,
   100→      gameAccent: game.accent,
   101→      productName: pItem.name,
   102→      priceLabel: pItem.priceLabel,
   103→    });
   104→    setSelectedItem({ item: pItem, category: categoryName });
   105→  };
   106→
   107→  const handleAddToCart = () => {
   108→    if (!selectedItem) return;
   109→    const success = cartAdd({
   110→      id: `${game.slug}-${selectedItem.item.id}`,
   111→      gameSlug: game.slug,
   112→      gameName: game.name,
   113→      gameEmoji: game.emoji,
   114→      productId: selectedItem.item.id,
   115→      productName: selectedItem.item.name,
   116→      priceLabel: selectedItem.item.priceLabel,
   117→      price: selectedItem.item.price,
   118→      category: selectedItem.category,
   119→    });

   121→      const isDuplicate = cartHas(`${game.slug}-${selectedItem.item.id}`);
   122→      // Close item modal FIRST, then show limit/duplicate message
   123→      setSelectedItem(null);
   124→      if (isDuplicate) {
   125→        toast({ title: "Item sudah ada di keranjang", variant: "destructive" });
   126→      } else {
   127→        // Show limit modal after item modal closes
   128→        setTimeout(() => setCartLimitModal(true), 100);
   129→      }
   130→      return;
   131→    }
   132→    setSelectedItem(null);
   133→    toast({ title: "Ditambahkan ke keranjang! 🛒", description: `${selectedItem.item.name} (${selectedItem.item.priceLabel})` });
   134→  };
   135→
   136→  const handleBuyNow = () => {
   137→    if (!selectedItem) return;
   138→    selectProduct({
   139→      gameSlug: game.slug,
   140→      gameName: game.name,
   141→      productId: selectedItem.item.id,
   142→      productName: selectedItem.item.name,
   143→      priceLabel: selectedItem.item.priceLabel,
   144→      price: selectedItem.item.price,
   145→      category: selectedItem.category,
   146→    });
   147→    setSelectedItem(null);
   148→    router.push("/checkout");
   149→  };
   150→
   151→  const _handlePick = (
   152→    item: (typeof game.categories)[number]["items"][number],
   153→    categoryName: string
   154→  ) => {
   155→    selectProduct({
   156→      gameSlug: game.slug,
   157→      gameName: game.name,
   158→      productId: item.id,
   159→      productName: item.name,
   160→      priceLabel: item.priceLabel,
   161→      price: item.price,
   162→      category: categoryName,
   163→    });
   164→    router.push("/checkout");
   165→  };
   166→
   167→  return (
   168→    <div className="relative">
   169→      {/* header band */}
   170→      <section className="relative overflow-hidden border-b-2 border-[#a020f0]/40 scanlines">
   171→        <div className="absolute inset-0 bg-grid opacity-50" />
   172→        <div
   173→          className="absolute -top-20 right-0 h-72 w-72 rounded-full blur-[100px]"
   174→          style={{ background: `${game.accent}33` }}
   175→        />
   176→        <div className="relative mx-auto max-w-7xl px-4 sm:px-6 py-10 sm:py-14">
   177→          <Link
   178→            href="/"
   179→            className="inline-flex items-center gap-2 font-pixel text-[9px] uppercase text-[#9a93a8] hover:text-[#c44bff] transition-colors"
   180→          >
   181→            <ArrowLeft className="size-3.5" /> Beranda
   182→          </Link>
   183→
   184→          <div className="mt-6 flex flex-col sm:flex-row items-start sm:items-center gap-5">
   185→            <div
   186→              className="flex h-20 w-20 sm:h-24 sm:w-24 items-center justify-center border-2 pixel-corner text-5xl sm:text-6xl"
   187→              style={{
   188→                background: "#121017",
   189→                borderColor: game.accent,
   190→                boxShadow: `0 0 22px ${game.accent}66`,
   191→              }}
   192→            >
   193→              {game.emoji}
   194→            </div>
   195→            <div className="flex-1">
   196→              <p className="font-pixel text-[9px] uppercase tracking-[0.3em] text-[#a020f0]">
   197→                STORE
   198→              </p>
   199→              <h1 className="mt-2 font-pixel text-2xl sm:text-4xl text-[#e5e5e5] text-glow-neon">
   200→                {game.name}
   201→              </h1>
   202→              <p className="mt-3 text-sm text-[#bcb4c9] max-w-xl">{game.description}</p>
   203→            </div>
   204→            <div className="hidden sm:block">
   205→              <div className="border-2 border-[#a020f0]/50 pixel-corner bg-[#a020f0]/5 px-4 py-3 text-center">
   206→                <p className="font-pixel text-[8px] uppercase text-[#9a93a8]">Total Joki</p>

   211→            </div>
   212→          </div>
   213→        </div>

   216→      {/* Notice banner (e.g. Blox Fruits CDK & SG warning) */}
   217→      {game.notice && (
   218→        <div className="mx-auto max-w-7xl px-4 sm:px-6 pt-8">

   221→              className={`relative overflow-hidden border-2 pixel-corner p-5 sm:p-6 ${
   222→                game.notice.type === "warning"
   223→                  ? "border-[#ff3b6b]/60 bg-[#ff3b6b]/10"
   224→                  : "border-[#7fd4ff]/50 bg-[#7fd4ff]/10"
   225→              }`}
   226→            >
   227→              <div className="flex items-start gap-3">
   228→                <span
   229→                  className={`mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center border-2 pixel-corner ${
   230→                    game.notice.type === "warning"
   231→                      ? "border-[#ff3b6b] text-[#ff3b6b]"
   232→                      : "border-[#7fd4ff] text-[#7fd4ff]"
   233→                  }`}
   234→                >
   235→                  <AlertTriangle className="size-4" />
   236→                </span>
   237→                <div className="min-w-0">

   239→                    className={`font-pixel text-[10px] sm:text-xs uppercase ${
   240→                      game.notice.type === "warning" ? "text-[#ff8aa3]" : "text-[#7fd4ff]"
   241→                    }`}
   242→                  >
   243→                    {game.notice.title}
   244→                  </p>
   245→                  <p className="mt-2 text-sm text-[#e5e5e5] leading-relaxed">{game.notice.body}</p>
   246→                </div>
   247→              </div>
   248→            </div>
   249→          </Reveal>
   250→        </div>
   251→      )}
   252→
   253→      {/* Advanced Filter Bar */}
   254→      <section className="mx-auto max-w-7xl px-4 sm:px-6 py-6">
   255→        <div className="glass-nav rounded-2xl p-4 sticky top-20 z-30">
   256→          <div className="flex flex-col sm:flex-row gap-3">
   257→            <input
   258→              type="text"
   259→              value={filterQuery}
   260→              onChange={(e) => setFilterQuery(e.target.value)}
   261→              placeholder="Cari item... (mis. level, raid, senjata)"
   262→              className="flex-1 bg-white/5 border border-white/10 rounded-xl px-4 py-2 text-sm text-zinc-100 placeholder:text-zinc-500 outline-none focus:border-violet-500/40"
   263→            />
   264→            <select
   265→              value={filterCategory}
   266→              onChange={(e) => setFilterCategory(e.target.value)}
   267→              className="bg-[#0a0a0a] border border-white/10 rounded-xl px-3 py-2 text-sm text-zinc-200 outline-none focus:border-violet-500/40"
   268→            >
   269→              <option value="all" className="bg-[#0a0a0a]">Semua Kategori</option>
   270→              {game.categories.map((c) => (
   271→                <option key={c.id} value={c.id} className="bg-[#0a0a0a]">{c.icon} {c.name}</option>
   272→              ))}
   273→            </select>
   274→            {allTags.length > 0 && (
   275→              <select
   276→                value={filterTag}
   277→                onChange={(e) => setFilterTag(e.target.value)}
   278→                className="bg-[#0a0a0a] border border-white/10 rounded-xl px-3 py-2 text-sm text-zinc-200 outline-none focus:border-violet-500/40"
   279→              >
   280→                <option value="all" className="bg-[#0a0a0a]">Semua Tag</option>
   281→                {allTags.map((t) => (
   282→                  <option key={t} value={t} className="bg-[#0a0a0a]">{t}</option>
   283→                ))}
   284→              </select>
   285→            )}
   286→            <select
   287→              value={sortBy}
   288→              onChange={(e) => setSortBy(e.target.value as any)}
   289→              className="bg-[#0a0a0a] border border-white/10 rounded-xl px-3 py-2 text-sm text-zinc-200 outline-none focus:border-violet-500/40"
   290→            >
   291→              <option value="default" className="bg-[#0a0a0a]">Urutkan: Default</option>
   292→              <option value="price-asc" className="bg-[#0a0a0a]">Harga: Rendah → Tinggi</option>
   293→              <option value="price-desc" className="bg-[#0a0a0a]">Harga: Tinggi → Rendah</option>
   294→              <option value="name" className="bg-[#0a0a0a]">Nama (A-Z)</option>
   295→            </select>
   296→            {(filterQuery || filterCategory !== "all" || filterTag !== "all" || sortBy !== "default") && (
   297→              <button
   298→                onClick={() => { setFilterQuery(""); setFilterCategory("all"); setFilterTag("all"); setSortBy("default"); }}
   299→                className="inline-flex items-center justify-center gap-1 rounded-xl bg-white/5 border border-white/10 px-3 py-2 text-xs text-zinc-400 hover:text-red-400 hover:border-red-500/30 transition-all"
   300→              >
   301→                <X className="size-3.5" /> Reset
   302→              </button>
   303→            )}
   304→          </div>
   305→        </div>
   306→      </section>
   307→
   308→      {/* category sections (filtered) */}
   309→      <section className="mx-auto max-w-7xl px-4 sm:px-6 py-8 sm:py-12 space-y-12">
   310→        {filteredCategories.length === 0 ? (
   311→          <div className="glass rounded-2xl p-12 text-center">
   312→            <p className="text-sm text-zinc-500">Tidak ada item yang cocok dengan filter.</p>
   313→            <button
   314→              onClick={() => { setFilterQuery(""); setFilterCategory("all"); setFilterTag("all"); setSortBy("default"); }}
   315→              className="mt-3 inline-flex items-center gap-2 rounded-xl bg-violet-500/10 border border-violet-500/20 px-4 py-2 text-sm text-violet-400 hover:bg-violet-500/20 transition-all"
   316→            >
   317→              Reset Filter
   318→            </button>
   319→          </div>
   320→        ) : (
   321→          filteredCategories.map((cat) => (
   322→          <div key={cat.id}>
   323→            <div className="flex items-center gap-3 mb-6">
   324→              <span className="text-2xl">{cat.icon}</span>
   325→              <h2 className="font-pixel text-sm sm:text-lg text-[#e5e5e5] text-glow-neon">
   326→                {cat.name}
   327→              </h2>
   328→              <div className="h-[2px] flex-1 bg-gradient-to-r from-[#a020f0] to-transparent" />
   329→              <span className="font-pixel text-[8px] uppercase text-[#9a93a8]">
   330→                {cat.items.length} item
   331→              </span>
   332→            </div>
   333→
   334→            <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
   335→              {cat.items.map((item, i) => {
   336→                const isSelected =
   337→                  order?.gameSlug === game.slug && order?.productId === item.id;
   338→                return (
   339→                  <Reveal key={item.id} delay={i * 60}>
   340→                    <ProductCard
   341→                      item={item}
   342→                      game={game}
   343→                      selected={!!isSelected}
   344→                      inCart={cartHas(`${game.slug}-${item.id}`)}
   345→                      onPick={() => handlePick(item, cat.name)}
   346→                    />
   347→                  </Reveal>
   348→                );
   349→              })}
   350→            </div>
   351→          </div>

   354→      </section>
   355→
   356→      {/* Reviews section */}
   357→      <StoreReviews game={game} />
   358→
   359→      {/* other games */}
   360→      <section className="mx-auto max-w-7xl px-4 sm:px-6 pb-16">
   361→        <div className="border-2 border-[#a020f0]/40 bg-[#121017]/60 pixel-corner p-6 sm:p-8">
   362→          <div className="flex items-center gap-2 mb-4">
   363→            <Tag className="size-4 text-[#c44bff]" />
   364→            <h3 className="font-pixel text-xs sm:text-sm text-[#e5e5e5] uppercase">
   365→              Jelajahi Game Lain

   373→                className="flex items-center gap-3 border-2 border-[#2a2436] hover:border-[#a020f0] pixel-corner px-4 py-3 text-left transition-colors group"
   374→              >
   375→                <span className="text-2xl">{g.emoji}</span>
   376→                <div className="flex-1 min-w-0">
   377→                  <p className="font-pixel text-[10px] text-[#e5e5e5] group-hover:text-[#c44bff] transition-colors truncate">
   378→                    {g.name}
   379→                  </p>

   395→            className="fixed inset-0 z-[9999] flex items-center justify-center p-4 bg-black/70 backdrop-blur-sm"
   396→            onClick={() => setSelectedItem(null)}

   408→                <div className="flex items-center gap-3">

   678→/* ============================ Store Reviews ============================ */
   679→function StoreReviews({ game }: { game: Game }) {
   680→  const allReviews = useReviews((s) => s.reviews);
   681→  const addReview = useReviews((s) => s.addReview);
   682→  const hydrated = useReviews((s) => s._hasHydrated);
   683→  const reviews = allReviews.filter((r) => r.gameSlug === game.slug);
   684→  const [name, setName] = useState("");
   685→  const [rating, setRating] = useState(5);
   686→  const [comment, setComment] = useState("");
   687→  const avgRating = reviews.length > 0 ? reviews.reduce((a, r) => a + r.rating, 0) / reviews.length : 0;
   688→
   689→  const handleSubmit = (e: React.FormEvent) => {
   690→    e.preventDefault();
   691→    if (!name.trim() || !comment.trim()) return;
   692→    addReview({ gameSlug: game.slug, gameName: game.name, productName: game.name, customerName: name.trim(), rating, comment: comment.trim() });
   693→    setName(""); setRating(5); setComment("");
   694→  };
   695→
   696→  if (!hydrated) return null;
   697→
   698→  return (
   699→    <section className="mx-auto max-w-7xl px-4 sm:px-6 py-12">
   700→      <div className="flex items-center gap-2 mb-6">
   701→        <Star className="size-5 text-amber-400" />
   702→        <h2 className="text-lg font-bold text-zinc-100">Review & Rating</h2>
   703→        {reviews.length > 0 && (
   704→          <span className="ml-2 inline-flex items-center gap-1 rounded-lg bg-amber-400/10 border border-amber-400/20 px-2 py-0.5 text-xs text-amber-400">
   705→            {avgRating.toFixed(1)} ★ · {reviews.length} review
   706→          </span>
   707→        )}
   708→      </div>
   709→      <div className="grid gap-6 lg:grid-cols-2">
   710→        <form onSubmit={handleSubmit} className="glass rounded-2xl p-5 space-y-3">
   711→          <h3 className="text-sm font-semibold text-zinc-300">Tulis Review</h3>
   712→          <input type="text" value={name} onChange={(e) => setName(e.target.value)} placeholder="Nama kamu" required
   713→            className="w-full bg-white/5 border border-white/10 rounded-xl px-3 py-2 text-sm text-zinc-100 outline-none focus:border-violet-500/40" />
   714→          <div className="flex items-center gap-2">
   715→            <span className="text-xs text-zinc-500">Rating:</span>
   716→            {[1,2,3,4,5].map((s) => (
   717→              <button key={s} type="button" onClick={() => setRating(s)} className="transition-transform hover:scale-110">
   718→                <Star className={cn("size-5", s <= rating ? "fill-amber-400 text-amber-400" : "text-zinc-600")} />
   719→              </button>
   720→            ))}
   721→          </div>
   722→          <textarea value={comment} onChange={(e) => setComment(e.target.value)} placeholder="Bagikan pengalaman joki kamu..." rows={3} required
   723→            className="w-full bg-white/5 border border-white/10 rounded-xl px-3 py-2 text-sm text-zinc-100 outline-none focus:border-violet-500/40 resize-none" />
   724→          <button type="submit" className="inline-flex items-center gap-2 rounded-xl bg-gradient-to-r from-violet-600 to-violet-500 px-4 py-2 text-sm text-white hover:from-violet-500 hover:to-violet-400 transition-all">
   725→            Kirim Review
   726→          </button>
   727→        </form>
   728→        <div className="space-y-3 max-h-80 overflow-y-auto">
   729→          {reviews.length === 0 ? (
   730→            <div className="glass rounded-2xl p-8 text-center">
   731→              <Star className="mx-auto size-8 text-zinc-700 mb-2" />
   732→              <p className="text-sm text-zinc-500">Belum ada review. Jadikan yang pertama!</p>
   733→            </div>
   734→          ) : (
   735→            reviews.map((r) => (
   736→              <div key={r.id} className="glass rounded-2xl p-4">
   737→                <div className="flex items-center justify-between mb-2">
   738→                  <span className="text-sm font-medium text-zinc-200">{r.customerName}</span>
   739→                  <div className="flex gap-0.5">{Array.from({length: 5}).map((_, i) => (
   740→                    <Star key={i} className={cn("size-3", i < r.rating ? "fill-amber-400 text-amber-400" : "text-zinc-700")} />
   741→                  ))}</div>
   742→                </div>
   743→                <p className="text-sm text-zinc-400 leading-relaxed">{r.comment}</p>
   744→                <p className="text-[10px] text-zinc-600 mt-2">{new Date(r.createdAt).toLocaleDateString("id-ID")}</p>
   745→              </div>

   747→          )}
   748→        </div>
   749→      </div>
   750→    </section>
