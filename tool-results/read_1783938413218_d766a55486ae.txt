   423→function Field({
   424→  label,
   425→  icon,
   426→  type,
   427→  placeholder,
   428→  value,
   429→  onChange,
   430→  toggleVisibility,
   431→}: {
   432→  label: string;
   433→  icon: React.ReactNode;
   434→  type: string;
   435→  placeholder: string;
   436→  value: string;
   437→  onChange: (e: React.ChangeEvent<HTMLInputElement>) => void;
   438→  /** Jika diisi, tampilkan tombol eye untuk show/hide password. */
   439→  toggleVisibility?: { show: boolean; onToggle: () => void };
   440→}) {
   441→  return (
   442→    <div>
   443→      <label className="font-pixel text-[9px] uppercase tracking-wide text-[#9a93a8] flex items-center gap-2">
   444→        <span className="text-[#c44bff]">{icon}</span>
   445→        {label}
   446→        {toggleVisibility && (
   447→          <span className="ml-auto font-pixel text-[7px] uppercase tracking-wide text-[#9a93a8]">
   448→            {toggleVisibility.show ? "Terlihat" : "Tersembunyi"}
   449→          </span>
   450→        )}
   451→      </label>
   452→      <div className="relative mt-2">
   453→        <input
   454→          type={type}
   455→          placeholder={placeholder}
